2. RECOMMENDED CMAKE INTEGRATION SNIPPET (e.g., cmake/s7_3d_gdk_title.cmake)
=============================================================================
if(WIN32)
    # Target 1: gdk_3d_scene (Pure standalone Xbox-ABI guest executable — NEVER links openscarlett_core!)
    add_executable(gdk_3d_scene
        tests/gdk_3d_scene/gdk_3d_scene.cpp
    )
    target_include_directories(gdk_3d_scene PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/gdk_3d_scene
    )
    target_compile_definitions(gdk_3d_scene PRIVATE
        WIN32_LEAN_AND_MEAN
        NOMINMAX
        _CRT_SECURE_NO_WARNINGS
    )
    target_link_libraries(gdk_3d_scene PRIVATE
        d3d12_xs
        xgameruntime
        GameInput
        XAudio2_9
        kernelx
        d3d12
        dxgi
        d3dcompiler
        user32
        kernel32
        ole32
        uuid
    )
    if(MSVC)
        target_compile_options(gdk_3d_scene PRIVATE /W4)
    else()
        target_compile_options(gdk_3d_scene PRIVATE -Wall -Wextra)
    endif()

    # Copy MicrosoftGame.Config alongside build output for direct portable/launcher packaging
    configure_file(
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/gdk_3d_scene/MicrosoftGame.Config
        ${CMAKE_BINARY_DIR}/gdk_3d_scene_MicrosoftGame.Config
        COPYONLY
    )

    # Target 2: test_gdk_3d_scene_dual_api (CTest dual-API isolated launch harness)
    openscarlett_add_test(test_gdk_3d_scene_dual_api tests/test_gdk_3d_scene_dual_api.cpp)
    add_dependencies(test_gdk_3d_scene_dual_api
        openscarlett
        openscarlett_hook
        d3d12_xs
        xgameruntime
        GameInput
        XAudio2_9
        kernelx
        gdk_3d_scene
    )
    set_tests_properties(test_gdk_3d_scene_dual_api PROPERTIES
        ENVIRONMENT "OPENSCARLETT_AUDIO_NULL=1"
    )
endif()
