3. THIN HOOK FOR src/main.cpp (--probe-unknown-imports)
--------------------------------------------------------------------------------
In `RunGuestExecutable(int argc, char** argv)`:

  a) Add flag variable:
       bool probe_unknown_imports = false;

  b) Inside the `for (int i = 2; i < argc; ++i)` option parser:
       } else if (arg == "--probe-unknown-imports") {
           probe_unknown_imports = true;

  c) Save/set/restore `OPENSCARLETT_PROBE_UNKNOWN_IMPORTS` around
     `openscarlett::LaunchWithApcHook(opts, res)` (identical pattern to
     `OPENSCARLETT_GPU_BACKEND`):
       bool had_prev_probe_env = false;
       std::string prev_probe_env;
       // Before LaunchWithApcHook:
       read_env_var("OPENSCARLETT_PROBE_UNKNOWN_IMPORTS", had_prev_probe_env, prev_probe_env);
       if (probe_unknown_imports) {
           SetEnvironmentVariableA("OPENSCARLETT_PROBE_UNKNOWN_IMPORTS", "1");
       }
       // After LaunchWithApcHook:
       if (probe_unknown_imports) {
           SetEnvironmentVariableA("OPENSCARLETT_PROBE_UNKNOWN_IMPORTS",
                                   had_prev_probe_env ? prev_probe_env.c_str() : nullptr);
       }

  d) In `PrintUsage()`, document:
       "      --probe-unknown-imports          Synthesize E_NOTIMPL trampolines & install VEH crash monitor\n"

4. THIN HOOK FOR src/loader/hook_dll.cpp (Probe Trampolines + VEH Monitor)
--------------------------------------------------------------------------------
Include:
    #include "openscarlett/loader/probe_veh.h"

Add static runtime state in anonymous namespace:
    openscarlett::MissingImportTrampolinePool g_probe_trampoline_pool;
    openscarlett::VehCrashMonitor g_veh_monitor;
    bool g_probe_unknown_imports = false;

In `InitializeHookRuntime(HMODULE self_mod)`:
  1. Read `g_probe_unknown_imports = openscarlett::IsProbeUnknownImportsEnabledByEnv();`
  2. When `g_probe_unknown_imports` is true:
       g_veh_monitor.SetTrampolinePool(&g_probe_trampoline_pool);
       if (!g_telemetry_path_utf8.empty()) {
           const std::string veh_json = g_telemetry_path_utf8 + ".veh_bringup.json";
           g_veh_monitor.SetAutoExportPath(veh_json, {openscarlett::WideToUtf8(g_shim_dir_w)});
       }
       (void)g_veh_monitor.Install(1);

In `Hook_GetProcAddress(HMODULE hModule, LPCSTR lpProcName)`:
  When `g_probe_unknown_imports` is true and real `GetProcAddress` returns
  `nullptr` for a tracked/guest module:
    const bool by_ord = (reinterpret_cast<uintptr_t>(lpProcName) <= 0xFFFFu);
    const uint16_t ord = by_ord ? static_cast<uint16_t>(reinterpret_cast<uintptr_t>(lpProcName)) : 0u;
    const std::string sym = by_ord ? std::string{} : std::string(lpProcName);
    if (void* stub = g_probe_trampoline_pool.GetOrCreateTrampoline(mod_name, sym, ord, by_ord)) {
        return reinterpret_cast<FARPROC>(stub);
    }

In `PatchModuleImports(HMODULE mod)`:
  When `g_probe_unknown_imports` is true, run
  `openscarlett::PatchPeImportsWithProbeTrampolines(mod, g_probe_trampoline_pool);`
  for non-system modules so unresolved static imports receive E_NOTIMPL stubs.

In `OpenScarlettHook_RecordCall(...)`:
  Also forward to `openscarlett::RecordProbeApiCall(module_name, symbol_name, 0, hr);`
  when `g_probe_unknown_imports` is true so the 50-entry VEH ring buffer captures
  all recent shim/D3D12 calls!

In `DllMain` (`DLL_PROCESS_DETACH`):
  Call `g_veh_monitor.Uninstall();` cleanly before shutdown.
