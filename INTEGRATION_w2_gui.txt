2. RECOMMENDED CMAKE INTEGRATION (FOR COORDINATOR MERGE)
================================================================================
Target 1: OpenScarlettLauncher (WIN32 executable)
  add_executable(OpenScarlettLauncher WIN32
      src/launcher/launcher_main.cpp
      src/launcher/updater.cpp
      src/launcher/hwtest_runner.cpp
  )
  target_include_directories(OpenScarlettLauncher PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/include
      ${CMAKE_CURRENT_SOURCE_DIR}/src/launcher
  )
  target_compile_definitions(OpenScarlettLauncher PRIVATE
      UNICODE _UNICODE
  )
  target_link_libraries(OpenScarlettLauncher PRIVATE
      openscarlett_core
      d2d1 dwrite dwmapi comctl32 comdlg32 shell32 ole32 uuid gdi32 user32
      winhttp bcrypt advapi32 version
  )
  if(MSVC)
      target_compile_options(OpenScarlettLauncher PRIVATE /W4 /WX)
  endif()
  add_dependencies(OpenScarlettLauncher openscarlett)

Target 2: test_launcher_updater (CTest executable)
  openscarlett_add_test(test_launcher_updater
      tests/test_launcher_updater.cpp
  )
  target_sources(test_launcher_updater PRIVATE
      src/launcher/updater.cpp
      src/launcher/hwtest_runner.cpp
  )
  target_link_libraries(test_launcher_updater PRIVATE
      winhttp bcrypt advapi32 version user32 shell32 ole32 uuid
  )
  if(MSVC)
      target_compile_options(test_launcher_updater PRIVATE /W4 /WX)
  endif()
  add_dependencies(test_launcher_updater openscarlett OpenScarlettLauncher)

Required Windows System Link Libraries (Zero Third-Party Dependencies):
  d2d1, dwrite, dwmapi, comctl32, comdlg32, shell32, ole32, uuid, gdi32,
  user32, winhttp, bcrypt, advapi32, version
